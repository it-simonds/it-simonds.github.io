# Autopilot-Bootstrap.ps1
# One-shot: register THIS device into Intune Autopilot from OOBE (no USB).
# Contains no secrets. Authorization = the admin sign-in via device code.
# Usage at OOBE (Shift+F10, then "powershell"):
#   irm https://<host>/Autopilot-Bootstrap.ps1 | iex

# Optional: only run from company egress IPs. Leave empty to skip the check.
$AllowedPublicIPs = @()      # e.g. @('203.0.113.10','203.0.113.11')
$DefaultGroupTag  = ''       # e.g. 'Corp-Laptop'; blank = no group tag

$ErrorActionPreference = 'Stop'
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
Set-ExecutionPolicy Bypass -Scope Process -Force

Write-Host "`n== Autopilot bootstrap ==" -ForegroundColor Cyan

# 1. Network gate (optional)
if ($AllowedPublicIPs.Count -gt 0) {
    $ip = (Invoke-RestMethod -Uri 'https://api.ipify.org' -TimeoutSec 10).Trim()
    if ($AllowedPublicIPs -notcontains $ip) {
        Write-Host "Public IP $ip is not on the allow list. Aborting." -ForegroundColor Red
        return
    }
    Write-Host "Network check passed ($ip)."
}

# 2. Show what we are about to register
$serial = (Get-CimInstance Win32_BIOS).SerialNumber
$model  = (Get-CimInstance Win32_ComputerSystem).Model
Write-Host "Device: $model, serial $serial"

$tag = Read-Host "Group tag (Enter for '$DefaultGroupTag')"
if ([string]::IsNullOrWhiteSpace($tag)) { $tag = $DefaultGroupTag }

$go = Read-Host "Register this device in Autopilot? (y/N)"
if ($go -notmatch '^(y|yes)$') { Write-Host 'Cancelled.'; return }

# 3. Prerequisites
Write-Host 'Installing prerequisites...'
Install-PackageProvider -Name NuGet -MinimumVersion 2.8.5.201 -Force | Out-Null
Install-Script -Name Get-WindowsAutopilotInfo -Force
$env:Path += ";$env:ProgramFiles\WindowsPowerShell\Scripts"

# 4. Sign in with device code (approve from another PC or phone), then upload
Write-Host 'Sign in: open the URL shown below on another device and enter the code.' -ForegroundColor Yellow
Import-Module Microsoft.Graph.Authentication -ErrorAction SilentlyContinue
Connect-MgGraph -UseDeviceCode -NoWelcome -Scopes 'DeviceManagementServiceConfig.ReadWrite.All'

$params = @{ Online = $true }
if ($tag) { $params.GroupTag = $tag }
Get-WindowsAutopilotInfo @params

Write-Host "`nDone. Check Intune > Devices > Enrollment > Windows Autopilot devices for serial $serial (can take a few minutes)." -ForegroundColor Green
